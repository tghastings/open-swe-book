% ---- Callout boxes: left rule, NO fill (Principle / Pitfall / Case study / ...) ----
% Grayscale only (interior prints B&W). A single thin vertical rule at the left
% marks the callout — no gray fill, no box — under a bold small-caps label. The
% type is named by the label; #1 (style key from callouts.lua) is reserved for
% future per-type variation but currently unused so every callout reads uniform.
\usepackage[most]{tcolorbox}
% unbreakable: a callout stays whole and moves to the next page rather than
% splitting mid-sentence at a page boundary (short admonitions read as artifacts
% when broken). All callouts here are far shorter than a text column.
\newtcolorbox{callout}[2]{%
  enhanced, unbreakable,
  colback=white, colframe=white, boxrule=0pt, boxsep=0pt,
  left=13pt, right=0pt, top=1pt, bottom=1pt,
  before skip=10pt, after skip=10pt,
  borderline west={1.6pt}{0pt}{black!72},
  fonttitle=\normalfont\bfseries\scshape, coltitle=black, colbacktitle=white,
  titlerule=0pt, bottomtitle=3pt, title={#2}}

% ---- Headings (scrbook, KOMA-native; grayscale) ----
% Chapter opener: SICP-style giant faded-gray numeral flush right, a hairline,
% then the title flush left. \IfArgIsEmpty gives \chapter* (Preface) a clean
% title-only head. \RedeclareSectionCommand{chapter} RESETS the vskip macros, so
% it MUST precede the two \renewcommand vskip lines below.
\setkomafont{chapter}{\normalfont\bfseries\huge}
\newcommand*{\chapnumfont}{\normalfont\bfseries\fontsize{60}{60}\selectfont\color{chapnum}}
\renewcommand*{\chapterformat}{\thechapter}
\renewcommand*{\raggedchapter}{\raggedright}
\RedeclareSectionCommand[beforeskip=0pt,afterskip=0pt,afterindent=false]{chapter}
\renewcommand*{\chapterheadstartvskip}{\vspace*{2.0\baselineskip}}
\renewcommand*{\chapterheadendvskip}{\vspace{2.6\baselineskip}}
\renewcommand*{\chapterlinesformat}[3]{%
  \IfArgIsEmpty{#2}{%
    \noindent #3%
  }{%
    \noindent\makebox[\linewidth][r]{\chapnumfont #2}\par\nobreak
    \vskip 0.15\baselineskip
    \noindent{\color{chaprule}\rule{\linewidth}{0.7pt}}\par\nobreak
    \vskip 0.95\baselineskip
    \noindent #3%
  }%
}

\RedeclareSectionCommand[beforeskip=2.0ex plus .8ex minus .2ex,afterskip=0.9ex plus .2ex,afterindent=false]{subsection}

\renewcommand*{\sectionlinesformat}[4]{%
  \Ifstr{#1}{section}{%
    \parbox[t]{\linewidth}{%
      \raggedsection\@hangfrom{\hskip #2#3}{#4}\par
      \vskip 0.35ex
      {\color{headrule}\rule{\linewidth}{0.4pt}}%
    }%
  }{%
    \@hangfrom{\hskip #2#3}{#4}%
  }%
}
\makeatother
